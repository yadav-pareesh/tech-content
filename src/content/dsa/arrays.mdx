## Two Pointers

<QuestionCard question="Two Sum — find two numbers that add up to a target" difficulty="easy" tags={["two pointers", "hash map", "O(n)"]}>

<CodeBlock language="javascript" code={`// Approach 1: Hash Map — O(n) time, O(n) space
function twoSum(nums, target) {
  const seen = new Map(); // { value: index }

  for (let i = 0; i < nums.length; i++) {
    const complement = target - nums[i];

    if (seen.has(complement)) {
      return [seen.get(complement), i];
    }

    seen.set(nums[i], i);
  }

  return [];
}

// Approach 2: Two Pointers (sorted array) — O(n) time, O(1) space
function twoSumSorted(sortedNums, target) {
  let left = 0;
  let right = sortedNums.length - 1;

  while (left < right) {
    const sum = sortedNums[left] + sortedNums[right];
    if (sum === target) return [left, right];
    if (sum < target) left++;
    else right--;
  }

  return [];
}

// Tests
twoSum([2, 7, 11, 15], 9);  // [0, 1]
twoSum([3, 2, 4], 6);       // [1, 2]`} />

**Time:** O(n) | **Space:** O(n) for hash map, O(1) for two pointers
</QuestionCard>

<QuestionCard question="Maximum Subarray (Kadane's Algorithm)" difficulty="medium" tags={["Kadane's", "dynamic programming", "subarray"]}>

Find the contiguous subarray with the largest sum.

<CodeBlock language="javascript" code={`function maxSubArray(nums) {
  let maxSum = nums[0];
  let currentSum = nums[0];

  for (let i = 1; i < nums.length; i++) {
    // Either extend current subarray or start fresh
    currentSum = Math.max(nums[i], currentSum + nums[i]);
    maxSum = Math.max(maxSum, currentSum);
  }

  return maxSum;
}

// Visual trace for [-2, 1, -3, 4, -1, 2, 1, -5, 4]:
// i=1: curr = max(1, -2+1=-1) = 1,  max = 1
// i=2: curr = max(-3, 1-3=-2) = -2, max = 1
// i=3: curr = max(4, -2+4=2) = 4,   max = 4
// i=4: curr = max(-1, 4-1=3) = 3,   max = 4
// i=5: curr = max(2, 3+2=5) = 5,    max = 5
// i=6: curr = max(1, 5+1=6) = 6,    max = 6  ← [4,-1,2,1]
// Answer: 6

maxSubArray([-2, 1, -3, 4, -1, 2, 1, -5, 4]); // 6`} />

**Time:** O(n) | **Space:** O(1)

<InterviewTip>
Kadane's insight: at each position, the maximum subarray ending here is either the current element alone (start fresh) or the current element plus the maximum subarray ending at the previous position.
</InterviewTip>
</QuestionCard>

## Sliding Window

<QuestionCard question="Longest Substring Without Repeating Characters" difficulty="medium" tags={["sliding window", "hash set", "string"]}>

<CodeBlock language="javascript" code={`function lengthOfLongestSubstring(s) {
  const charSet = new Set();
  let left = 0;
  let maxLen = 0;

  for (let right = 0; right < s.length; right++) {
    // Shrink window until duplicate is removed
    while (charSet.has(s[right])) {
      charSet.delete(s[left]);
      left++;
    }

    charSet.add(s[right]);
    maxLen = Math.max(maxLen, right - left + 1);
  }

  return maxLen;
}

// Trace for "abcabcbb":
// r=0: a → {a}, len=1
// r=1: b → {a,b}, len=2
// r=2: c → {a,b,c}, len=3
// r=3: a → remove a: {b,c,a}, len=3
// r=4: b → remove b: {c,a,b}, len=3
// r=5: c → remove c: {a,b,c}, len=3
// r=6: b → remove a,b: {c,b}, len=3
// r=7: b → remove c,b: {b}, len=3

lengthOfLongestSubstring("abcabcbb"); // 3
lengthOfLongestSubstring("bbbbb");    // 1
lengthOfLongestSubstring("pwwkew");   // 3`} />

**Time:** O(n) | **Space:** O(min(m, n)) where m is charset size
</QuestionCard>

## Prefix Sum

<QuestionCard question="Subarray Sum Equals K" difficulty="medium" tags={["prefix sum", "hash map", "counting"]}>

Count subarrays with sum equal to k.

<CodeBlock language="javascript" code={`function subarraySum(nums, k) {
  const prefixCounts = new Map([[0, 1]]); // empty subarray has sum 0
  let prefixSum = 0;
  let count = 0;

  for (const num of nums) {
    prefixSum += num;

    // If (prefixSum - k) exists in our map,
    // there's a subarray ending here with sum k
    const needed = prefixSum - k;
    if (prefixCounts.has(needed)) {
      count += prefixCounts.get(needed);
    }

    prefixCounts.set(prefixSum, (prefixCounts.get(prefixSum) ?? 0) + 1);
  }

  return count;
}

// Key insight: sum(i, j) = prefixSum[j] - prefixSum[i-1]
// If sum(i, j) = k, then prefixSum[j] - k = prefixSum[i-1]

subarraySum([1, 1, 1], 2); // 2 → [1,1] appears at indices (0,1) and (1,2)
subarraySum([1, 2, 3], 3); // 2 → [3] and [1,2]`} />

**Time:** O(n) | **Space:** O(n)
</QuestionCard>

## Binary Search

<QuestionCard question="Binary search and its variations" difficulty="medium" tags={["binary search", "O(log n)", "sorted array"]}>

<CodeBlock language="javascript" code={`// Standard binary search
function binarySearch(nums, target) {
  let left = 0;
  let right = nums.length - 1;

  while (left <= right) {
    const mid = left + Math.floor((right - left) / 2); // avoid overflow
    if (nums[mid] === target) return mid;
    if (nums[mid] < target) left = mid + 1;
    else right = mid - 1;
  }

  return -1;
}

// Find leftmost (first occurrence)
function findFirst(nums, target) {
  let left = 0, right = nums.length - 1, result = -1;
  while (left <= right) {
    const mid = left + Math.floor((right - left) / 2);
    if (nums[mid] === target) {
      result = mid;
      right = mid - 1; // keep searching left
    } else if (nums[mid] < target) {
      left = mid + 1;
    } else {
      right = mid - 1;
    }
  }
  return result;
}

// Search in rotated sorted array
function searchRotated(nums, target) {
  let left = 0, right = nums.length - 1;
  while (left <= right) {
    const mid = left + Math.floor((right - left) / 2);
    if (nums[mid] === target) return mid;

    // Left half is sorted
    if (nums[left] <= nums[mid]) {
      if (nums[left] <= target && target < nums[mid]) right = mid - 1;
      else left = mid + 1;
    } else { // Right half is sorted
      if (nums[mid] < target && target <= nums[right]) left = mid + 1;
      else right = mid - 1;
    }
  }
  return -1;
}`} />

**Time:** O(log n) | **Space:** O(1)
</QuestionCard>

## Common Array Complexity

| Operation | Array | Sorted Array |
|-----------|-------|-------------|
| Access by index | O(1) | O(1) |
| Search | O(n) | O(log n) binary search |
| Insert at end | O(1) amortized | O(n) to maintain order |
| Insert at start/middle | O(n) | O(n) |
| Delete | O(n) | O(n) |

## Coding Practice: Array Problems

<QuestionCard question="Find duplicate elements in an array" difficulty="easy" tags={["duplicates", "Set", "hash map", "coding practice"]}>

<CodeBlock language="javascript" code={`// Approach: Track seen elements with a Set — O(n) time, O(n) space
function findDuplicates(nums) {
  const seen = new Set();
  const duplicates = new Set();

  for (const num of nums) {
    if (seen.has(num)) {
      duplicates.add(num);
    } else {
      seen.add(num);
    }
  }

  return [...duplicates];
}

findDuplicates([1, 2, 3, 2, 4, 5, 1, 2]); // [2, 1]`} />
</QuestionCard>

<QuestionCard question="Remove duplicates from an array" difficulty="easy" tags={["deduplicate", "Set", "filter", "coding practice"]}>

<CodeBlock language="javascript" code={`// Approach 1: Set — O(n) time, O(n) space (cleanest)
function removeDuplicates(nums) {
  return [...new Set(nums)];
}

// Approach 2: Without Set (using a plain object/Map or filter)
function removeDuplicatesManual(nums) {
  const seen = {};
  const result = [];

  for (const num of nums) {
    if (!seen[num]) {
      seen[num] = true;
      result.push(num);
    }
  }
  return result;
}

removeDuplicates([1, 2, 2, 3, 4, 4, 5]); // [1, 2, 3, 4, 5]`} />
</QuestionCard>

<QuestionCard question="Find the second-largest number in an array" difficulty="easy" tags={["second largest", "single pass", "O(n)", "coding practice"]}>

<CodeBlock language="javascript" code={`// Single pass — O(n) time, O(1) space (better than sorting O(n log n))
function secondLargest(nums) {
  if (nums.length < 2) return null;

  let first = -Infinity;
  let second = -Infinity;

  for (const num of nums) {
    if (num > first) {
      second = first;
      first = num;
    } else if (num > second && num !== first) {
      second = num;
    }
  }

  return second === -Infinity ? null : second;
}

secondLargest([10, 5, 20, 8, 20]); // 10 (handles duplicate max correctly)
secondLargest([5, 5, 5]);          // null`} />

<InterviewTip>
Always clarify edge cases: What if the array has duplicates of the largest number (`[20, 20, 10]`)? What if all elements are equal (`[5, 5]`)? Checking `num !== first` handles duplicate maximums cleanly in `O(n)` time.
</InterviewTip>
</QuestionCard>

<QuestionCard question="Flatten a nested array" difficulty="medium" tags={["recursion", "flat", "stack", "coding practice"]}>

<CodeBlock language="javascript" code={`// Approach 1: Built-in Array.prototype.flat(Infinity)
const flatBuiltIn = [1, [2, [3, [4]], 5]].flat(Infinity); // [1, 2, 3, 4, 5]

// Approach 2: Recursive implementation (expected in interviews!)
function flattenArray(arr) {
  const result = [];

  for (const item of arr) {
    if (Array.isArray(item)) {
      result.push(...flattenArray(item));
    } else {
      result.push(item);
    }
  }

  return result;
}

// Approach 3: Iterative with a stack (avoids call stack overflow on deeply nested arrays)
function flattenIterative(arr) {
  const stack = [...arr];
  const result = [];

  while (stack.length > 0) {
    const next = stack.pop();
    if (Array.isArray(next)) {
      stack.push(...next);
    } else {
      result.push(next);
    }
  }

  return result.reverse();
}

flattenArray([1, [2, [3, 4], 5], 6]); // [1, 2, 3, 4, 5, 6]`} />
</QuestionCard>
