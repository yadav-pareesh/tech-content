## Types vs Interfaces

<QuestionCard question="What is the difference between type and interface in TypeScript?" difficulty="medium" tags={["type", "interface", "extends", "intersection"]}>

Both `type` and `interface` can describe object shapes, but they have key differences.

<CodeBlock language="typescript" code={`// Interface — can be extended and merged
interface User {
  id: number;
  name: string;
}

interface Admin extends User {
  role: 'admin';
  permissions: string[];
}

// Declaration merging (only interfaces can do this)
interface Window {
  analytics: Analytics; // augments the global Window type
}

// Type aliases — more flexible
type UserId = number;
type Status = 'active' | 'inactive' | 'suspended';

// Intersection types (like extends for types)
type AdminUser = User & {
  role: 'admin';
  permissions: string[];
};

// Types can represent unions; interfaces cannot
type StringOrNumber = string | number;
// ❌ interface StringOrNumber = string | number; (not valid)

// When to use each:
// - Interface: when defining object shapes that might be extended or augmented
// - Type: for unions, intersections, primitives, tuples, complex mapped types`} />

| Feature | `interface` | `type` |
|---------|------------|--------|
| Extends | ✅ `extends` | ✅ `&` intersection |
| Unions | ❌ | ✅ |
| Primitives | ❌ | ✅ |
| Declaration merging | ✅ | ❌ |
| Computed properties | ❌ | ✅ |
</QuestionCard>

## Generics

<QuestionCard question="Explain generics in TypeScript with practical examples" difficulty="medium" tags={["generics", "type parameters", "constraints"]}>

Generics allow you to write **reusable, type-safe code** that works with different types while preserving type information.

<CodeBlock language="typescript" code={`// Basic generic function
function identity<T>(value: T): T {
  return value;
}

identity<string>('hello'); // string
identity(42);              // TypeScript infers: number

// Generic with constraints
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key];
}

const user = { id: 1, name: 'Alice' };
getProperty(user, 'name');  // ✓ string
getProperty(user, 'email'); // ✗ TypeScript error — 'email' not in user

// Generic React component
interface ListProps<T> {
  items: T[];
  renderItem: (item: T, index: number) => React.ReactNode;
  keyExtractor: (item: T) => string;
}

function TypedList<T>({ items, renderItem, keyExtractor }: ListProps<T>) {
  return (
    <ul>
      {items.map((item, i) => (
        <li key={keyExtractor(item)}>{renderItem(item, i)}</li>
      ))}
    </ul>
  );
}

// Generic API response wrapper
type ApiResponse<T> = {
  data: T;
  error: string | null;
  loading: boolean;
};

type UserResponse = ApiResponse<User>;
type PostsResponse = ApiResponse<Post[]>;`} />
</QuestionCard>

## Union and Intersection Types

<QuestionCard question="How do union types and type narrowing work?" difficulty="medium" tags={["union types", "type narrowing", "type guards", "discriminated union"]}>

<CodeBlock language="typescript" code={`// Union type — value can be one of several types
type StringOrNumber = string | number;

function formatId(id: StringOrNumber): string {
  // Type narrowing with typeof
  if (typeof id === 'string') {
    return id.toUpperCase(); // id: string here
  }
  return id.toFixed(2);     // id: number here
}

// Discriminated unions — the right way to model variants
type Shape =
  | { kind: 'circle'; radius: number }
  | { kind: 'rectangle'; width: number; height: number }
  | { kind: 'triangle'; base: number; height: number };

function area(shape: Shape): number {
  switch (shape.kind) {
    case 'circle':
      return Math.PI * shape.radius ** 2; // shape: { kind: 'circle', radius }
    case 'rectangle':
      return shape.width * shape.height;
    case 'triangle':
      return 0.5 * shape.base * shape.height;
    // TypeScript warns if you miss a case!
  }
}

// User-defined type guard
interface Cat { meow(): void; type: 'cat'; }
interface Dog { bark(): void; type: 'dog'; }

function isCat(animal: Cat | Dog): animal is Cat {
  return animal.type === 'cat';
}

function handleAnimal(animal: Cat | Dog) {
  if (isCat(animal)) {
    animal.meow(); // animal: Cat
  } else {
    animal.bark(); // animal: Dog
  }
}`} />
</QuestionCard>

## Utility Types

<QuestionCard question="What are the most important TypeScript utility types?" difficulty="medium" tags={["Partial", "Required", "Pick", "Omit", "Record", "Readonly"]}>

<CodeBlock language="typescript" code={`interface User {
  id: number;
  name: string;
  email: string;
  age?: number;
}

// Partial<T> — all properties optional
type UserUpdate = Partial<User>;
// { id?: number; name?: string; email?: string; age?: number }

// Required<T> — all properties required
type FullUser = Required<User>;
// { id: number; name: string; email: string; age: number }

// Pick<T, K> — select subset of properties
type UserPreview = Pick<User, 'id' | 'name'>;
// { id: number; name: string }

// Omit<T, K> — exclude properties
type UserWithoutId = Omit<User, 'id'>;
// { name: string; email: string; age?: number }

// Record<K, V> — map type
type UserMap = Record<string, User>;
// { [key: string]: User }

// Readonly<T> — all properties immutable
type ImmutableUser = Readonly<User>;

// ReturnType<T> — extract return type of a function
function getUser() { return { id: 1, name: 'Alice' }; }
type GetUserReturn = ReturnType<typeof getUser>;
// { id: number; name: string }

// Extract and Exclude for union types
type A = 'a' | 'b' | 'c';
type B = Extract<A, 'a' | 'b'>; // 'a' | 'b'
type C = Exclude<A, 'a' | 'b'>; // 'c'

// NonNullable — removes null and undefined
type MaybeString = string | null | undefined;
type DefiniteString = NonNullable<MaybeString>; // string`} />

<InterviewTip>
Demonstrate using utility types in real scenarios: `Partial<T>` for update endpoints, `Pick<T>` for view models, `Record<string, T>` for lookup maps. This shows practical experience beyond just memorizing the API.
</InterviewTip>
</QuestionCard>

## never and unknown

<QuestionCard question="When and why do you use never and unknown in TypeScript?" difficulty="hard" tags={["never", "unknown", "type safety", "exhaustive checks"]}>

<CodeBlock language="typescript" code={`// unknown — safe alternative to any
// Values of type 'unknown' require type narrowing before use
function parseData(data: unknown) {
  // Can't use 'data' directly:
  // data.trim(); ❌ TypeScript error

  if (typeof data === 'string') {
    return data.trim(); // ✅ narrowed to string
  }
  if (Array.isArray(data)) {
    return data.length; // ✅ narrowed to array
  }
  throw new Error('Unexpected data type');
}

// never — represents values that never occur
// Use in exhaustive type checking
type Direction = 'north' | 'south' | 'east' | 'west';

function move(dir: Direction): void {
  switch (dir) {
    case 'north': return;
    case 'south': return;
    case 'east': return;
    case 'west': return;
    default:
      // If you add a new Direction without handling it,
      // TypeScript will error here because 'dir' won't be 'never'
      const exhausted: never = dir;
      throw new Error(\`Unhandled direction: \${exhausted}\`);
  }
}

// never as function return type (functions that never return)
function throwError(message: string): never {
  throw new Error(message);
}

function infiniteLoop(): never {
  while (true) {}
}`} />
</QuestionCard>
